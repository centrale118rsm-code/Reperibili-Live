# ═══════════════════════════════════════════════════════════════════════════
# REPERIBILI LIVE come SFONDO DEL DESKTOP
# Nessun programma da installare, nessun permesso di amministratore, nessun file .ps1:
# questo testo viene letto ed eseguito da PowerShell. Ogni minuto Edge, invisibile,
# "fotografa" la pagina del monitor e l'immagine diventa lo sfondo di Windows.
# Modalità ($RLMode): installa | run | ferma | disinstalla
# ═══════════════════════════════════════════════════════════════════════════
# ── Impostazioni ──
# Parametri utili nell'indirizzo: margine=9 (fascia libera a sinistra per le icone, % della larghezza)
# tema=notte|chiaro  turni=0..6  rubrica=1|0  orologio=1|0  reperibili=1|0  avviso=1|0  minuti=30
$Url        = 'https://centrale118rsm-code.github.io/Reperibili-Live/?sfondo=1&margine=9'
$Intervallo = 60        # secondi fra un aggiornamento e l'altro (60 = ogni minuto, allineato all'orologio)
$AttesaDati = 20000     # millisecondi lasciati alla pagina per caricare turni e reperibili

$ErrorActionPreference = 'Continue'
if(-not $RLMode){ $RLMode = 'run' }
$NomeTesto = 'ReperibiliLive-Sfondo.txt'
$avvio = Join-Path $env:APPDATA 'Microsoft\Windows\Start Menu\Programs\Startup\ReperibiliLive-Sfondo.cmd'
$candidate = @((Join-Path $env:LOCALAPPDATA 'ReperibiliLive'), (Join-Path ([Environment]::GetFolderPath('MyDocuments')) 'ReperibiliLive'), (Join-Path $env:USERPROFILE 'ReperibiliLive'))

function Cartella-Esistente { foreach($c in $candidate){ if(Test-Path (Join-Path $c $NomeTesto)){ return $c } }; return $null }
$dir = Cartella-Esistente
if(-not $dir){ $dir = $candidate[0] }
$log = Join-Path $dir 'registro.txt'
function Log($m){ try { Add-Content -Path $log -Value ("{0:yyyy-MM-dd HH:mm:ss}  {1}" -f (Get-Date), $m) } catch {} }

function Ferma-Altri {
  try {
    Get-CimInstance Win32_Process -Filter "Name='powershell.exe'" |
      Where-Object { $_.ProcessId -ne $PID -and $_.CommandLine -like "*RLMode='run'*" } |
      ForEach-Object { Stop-Process -Id $_.ProcessId -Force -ErrorAction SilentlyContinue }
  } catch {}
}
$wpOk = $false
try {
  Add-Type -ErrorAction Stop -TypeDefinition @"
using System; using System.Runtime.InteropServices;
public static class RLWin {
  [DllImport("user32.dll", CharSet = CharSet.Unicode, SetLastError = true)] public static extern bool SystemParametersInfo(int a, int b, string c, int d);
  [DllImport("user32.dll")] public static extern bool SetProcessDPIAware();
  [DllImport("user32.dll")] public static extern int GetSystemMetrics(int i);
}
"@
  $wpOk = $true
} catch { }
function Imposta-Sfondo($path){
  try { Set-ItemProperty 'HKCU:\Control Panel\Desktop' -Name WallPaper -Value $path } catch {}
  if($wpOk){ return [RLWin]::SystemParametersInfo(20, 0, $path, 3) }
  for($k = 0; $k -lt 3; $k++){ Start-Process rundll32.exe -ArgumentList 'user32.dll,UpdatePerUserSystemParameters 1, True' -WindowStyle Hidden -Wait }
  return $true
}
function Risoluzione {
  if($wpOk){ try { [void][RLWin]::SetProcessDPIAware(); return @([RLWin]::GetSystemMetrics(0), [RLWin]::GetSystemMetrics(1)) } catch {} }
  try { $v = Get-CimInstance Win32_VideoController | Where-Object { $_.CurrentHorizontalResolution } | Select-Object -First 1; if($v){ return @([int]$v.CurrentHorizontalResolution, [int]$v.CurrentVerticalResolution) } } catch {}
  return @(1920, 1080)
}

# ── INSTALLA: copia questo testo in una cartella permessa, avvio automatico, partenza ──
if($RLMode -eq 'installa'){
  Ferma-Altri
  $scelta = $null
  foreach($c in $candidate){
    try {
      New-Item -ItemType Directory -Force -Path $c -ErrorAction Stop | Out-Null
      $f = Join-Path $c $NomeTesto
      Set-Content -Path $f -Value $RLText -Encoding UTF8 -ErrorAction Stop
      $scelta = $c; break
    } catch { Write-Host ("  Cartella non permessa: " + $c) }
  }
  if(-not $scelta){ Write-Host ''; Write-Host '  Nessuna cartella permessa per salvare lo script: il PC lo blocca.'; exit 3 }
  $dir = $scelta; $log = Join-Path $dir 'registro.txt'
  $txt = Join-Path $dir $NomeTesto
  $cmdRun = "-NoProfile -WindowStyle Hidden -ExecutionPolicy Bypass -Command `"`$RLMode='run'; iex (Get-Content -Raw -Encoding UTF8 '$txt')`""
  try { Set-Content -Path $avvio -Value ('@start "" /min powershell ' + $cmdRun) -Encoding ASCII -ErrorAction Stop }
  catch { Write-Host '  Avvio automatico non creato (cartella Esecuzione automatica bloccata): andra'' lanciato a mano.' }
  Start-Process powershell -ArgumentList $cmdRun -WindowStyle Hidden
  Log "Installato in $dir"
  Write-Host ("  Script salvato in: " + $dir)
  exit 0
}
if($RLMode -eq 'ferma'){ Ferma-Altri; exit 0 }
if($RLMode -eq 'disinstalla'){
  Ferma-Altri
  Remove-Item $avvio -Force -ErrorAction SilentlyContinue
  $orig = Join-Path $dir 'sfondo-originale.txt'
  if(Test-Path $orig){ $p = (Get-Content $orig -Raw).Trim(); if($p -and (Test-Path $p)){ [void](Imposta-Sfondo $p) } }
  Log 'Disinstallato'
  exit 0
}

# ── RUN: ciclo di aggiornamento dello sfondo ──
# una sola copia alla volta
try {
  $altri = Get-CimInstance Win32_Process -Filter "Name='powershell.exe'" | Where-Object { $_.ProcessId -ne $PID -and $_.CommandLine -like "*RLMode='run'*" }
  if($altri){ exit 0 }
} catch {}
if((Test-Path $log) -and (Get-Item $log).Length -gt 1MB){ Remove-Item $log -Force -ErrorAction SilentlyContinue }
$profilo = Join-Path $dir 'edge-profilo'
New-Item -ItemType Directory -Force -Path $profilo | Out-Null
$edge = @("${env:ProgramFiles(x86)}\Microsoft\Edge\Application\msedge.exe",
          "$env:ProgramFiles\Microsoft\Edge\Application\msedge.exe",
          "$env:LOCALAPPDATA\Microsoft\Edge\Application\msedge.exe") | Where-Object { $_ -and (Test-Path $_) } | Select-Object -First 1
if(-not $edge){ Log 'Microsoft Edge non trovato'; exit 1 }
# sfondo attuale, per rimetterlo con Disinstalla-Sfondo.cmd
$orig = Join-Path $dir 'sfondo-originale.txt'
if(-not (Test-Path $orig)){ try { (Get-ItemProperty 'HKCU:\Control Panel\Desktop').WallPaper | Set-Content -Path $orig -Encoding UTF8 } catch {} }
try {
  Set-ItemProperty 'HKCU:\Control Panel\Desktop' -Name WallpaperStyle -Value '10'   # riempi
  Set-ItemProperty 'HKCU:\Control Panel\Desktop' -Name TileWallpaper -Value '0'
} catch {}
Log ("Avvio con $edge" + $(if($wpOk){''}else{' (modalità ridotta)'}))
$n = 0
while($true){
  try {
    $r = Risoluzione; $w = $r[0]; $h = $r[1]
    $file = Join-Path $dir ("sfondo-{0}.png" -f ($n % 2))     # due nomi alternati: Windows ricarica sempre l'immagine
    $tmp  = Join-Path $dir 'scatto.png'
    Remove-Item $tmp -Force -ErrorAction SilentlyContinue
    $sep = '?'; if($Url -match '\?'){ $sep = '&' }
    $u = $Url + $sep + 't=' + (Get-Date -Format 'yyyyMMddHHmmss')
    $argv = "--headless --disable-gpu --hide-scrollbars --mute-audio --no-first-run --no-default-browser-check " +
            "--user-data-dir=`"$profilo`" --window-size=$w,$h --force-device-scale-factor=1 " +
            "--virtual-time-budget=$AttesaDati --screenshot=`"$tmp`" `"$u`""
    $p = Start-Process -FilePath $edge -ArgumentList $argv -WindowStyle Hidden -PassThru
    Wait-Process -Id $p.Id -Timeout 90 -ErrorAction SilentlyContinue
    if(Get-Process -Id $p.Id -ErrorAction SilentlyContinue){ Stop-Process -Id $p.Id -Force -ErrorAction SilentlyContinue; Log 'Edge non ha risposto in tempo' }
    if((Test-Path $tmp) -and (Get-Item $tmp).Length -gt 15000){
      Move-Item $tmp $file -Force
      if(Imposta-Sfondo $file){ $n++ } else { Log 'Windows ha rifiutato lo sfondo (forse bloccato dai criteri aziendali)' }
    } else { Log 'Nessuna immagine prodotta (Edge invisibile bloccato o pagina non raggiungibile)' }
  } catch { Log ('Errore: ' + $_.Exception.Message) }
  if($Intervallo -eq 60){
    # riparte un attimo dopo l'inizio del minuto, così l'ora sullo sfondo è quella giusta
    $now = Get-Date
    $ms = (60 - $now.Second) * 1000 - $now.Millisecond + 300
    if($ms -lt 1000){ $ms = 1000 }
    Start-Sleep -Milliseconds $ms
  } else { if($Intervallo -lt 15){ $Intervallo = 15 }; Start-Sleep -Seconds $Intervallo }
}
